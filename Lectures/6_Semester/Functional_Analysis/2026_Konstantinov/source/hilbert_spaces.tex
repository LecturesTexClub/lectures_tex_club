\chapter{Геометрия гильбертовых пространств.}
\begin{definition}
    Пусть $V$ -- комплексное векторное пространство.
    Будем называть его пространством с внутренним произведением (или -- евклидовым пространством), если существует функция $(\cdot, \cdot)\colon V \times V \to \Cm$ такая, что $\forall x, y, z \in V$ и $\forall \alpha \in \Cm\colon$
    \begin{itemize}
        \item $(x, x) \geq 0$ и $(x, x) = 0 \Lra x = 0$;
        \item $(x, y + z) = (x, y) + (x, z)$;
        \item $(x, \alpha y) = \alpha(x, y)$;
        \item $(x, y) = \overline{(y, x)}$.
    \end{itemize}
\end{definition}
