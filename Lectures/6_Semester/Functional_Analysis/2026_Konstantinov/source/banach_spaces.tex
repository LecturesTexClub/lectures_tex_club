\part{Банаховы пространства. Введение.}
\chapter{Определения и примеры.}
\begin{definition}
    Полное нормированное линейное пространство называется банаховым.
\end{definition}
\begin{example}
    Пространства $L_p(X, \MM, \mu)$ (где $(X, \MM, \mu)$ -- пространство с мерой) являются банаховыми при $p \in [1, +\infty]$.
\end{example}
