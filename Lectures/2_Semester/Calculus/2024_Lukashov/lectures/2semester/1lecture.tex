\section{Дифференциальное исчисление функций многих переменных}

\subsection{Предел и непрерывность функций многих переменных}

\begin{definition}
	\textit{Функцией многих переменных} будем называть $f: D \to \R,\ D \subset \R^n$
\end{definition}

\begin{definition}
	\textit{(Предел функции по множеству $D$) (Конкретизация определения из первого семестра)(речь идёт о конечных пределах)} Если $\vv{x}_0$ - предельная точка множества $D$, то \textit{предел функции многих переменных} можно записать следующим образом:
	\[
		\liml_{\substack{\vv{x} \to \vv{x}_0 \\ \vv{x} \in D}} f(\vv{x}) = l
	\]
	Или же в другом виде:
	\[
		\liml_{\substack{(x_1, \ldots, x_n) \to (x_1^{(0)}, \ldots, x_n^{(0)}) \\ (x_1, \ldots, x_n) \in D}} f(x_1, \ldots, x_n) = l
	\]
	
	Теперь для функции многих переменных $f: D \to \R,\ D \subset \R^n$ предел расписывается следующим образом:
	\begin{itemize}
		\item По Коши:
		\[
			\liml_{\substack{\vv{x} \to \vv{x}_0 \\ \vv{x} \in D}} f(\vv{x}) = l \lra (\forall \eps > 0)\ (\exists \delta > 0) : (\forall \vv{x} \in D,\ 0 < |\vv{x} - \vv{x}_0| < \delta\ )\hookrightarrow |f(\vv{x}) - l| < \eps
		\]
		
		\item По Гейне:
		\[
			\liml_{\substack{\vv{x} \to \vv{x}_0 \\ \vv{x} \in D}} f(\vv{x}) = l \lra \left(\forall \{\vv{x}_k\}_{k = 1}^\infty \subset D \bs \{\vv{x}_0\},\ \liml_{k \to \infty} \vv{x}_k = \vv{x_0}\right) \hookrightarrow \liml_{k \to \infty} f(\vv{x}_k) = l
		\]
	\end{itemize}
\end{definition}

\begin{note}
	Почему $\vv{x}_0$ - обязательно предельная точка? Проблема в том, что если мы откажемся от такого предположения, то точка $\vv{x}_0$ может оказаться изолированной. Значит, для каких-то $\eps > 0$ часть определения про $\forall \vv{x} \in D,\ 0 < |\vv{x} - \vv{x}_0| < \delta$ окажется всегда верной. Из этого может получиться, что у нас не будет единственности предела и некоторые другие плохие ситуации.
\end{note}

\begin{proposition}
	Определения по Коши и по Гейне эквивалентны. (см. теорию пределов в метрических пространствах)
\end{proposition}

\begin{definition}
	\textit{(Предел по совокупности переменных)}
	Если $f$ определена в в какой-то проколотой окрестности точки $\vv{x}_0$, то в пределе по множеству $D$ можно опустить принадлежность $\vv{x}$ к $D$. Тогда предел записывается так:
	\[
		\liml_{\vv{x} \to \vv{x}_0} f(\vv{x}) = l
	\]
	Также его называют двойным или тройным, если речь идет о плоскости и пространстве соответственно.
\end{definition}
\begin{definition} (Предел по направлению)
	Если $f$ определена в в какой-то проколотой окрестности точки $\vv{x}_0$, \textit{пределом по направлению} $\vv{d} \in \R^n \bs \{\vv{0}\}$ в точке $\vv{x}_0$, называется предел следующего вида:
	\[
		\liml_{\substack{\vv{x} \to \vv{x}_0 \\ \vv{x} \in D_{\vv{x}_0, \vv{d}}}} f(\vv{x}) = l
	\]
	где	\(
	D_{\vv{x}_0, \vv{e}} = D \cap \{\vv{x} \in \R^n \such \vv{x} - \vv{x}_0 = \vv{d} \cdot t,\ t > 0\}
	\)
\end{definition}

\begin{anote}
	Два года назад в определении предела по направлению брали прямую, а не луч как сейчас. Лукашов говорит, что аргументы есть за оба типа определения, но для практического применения удобнее через луч.
\end{anote}

\begin{lemma}
	Если $f$ определена в некоторой проколотой окрестности точки $\vv{x}_0$, то
	\[
		\liml_{\substack{\vv{x} \to \vv{x}_0 \\ \vv{x} \in D_{\vv{x}_0, \vv{d}}}} f(\vv{x}) = \liml_{t \to +0} f(\vv{x}_0 + t \vv{d})
	\]
	(равенство надо понимать, что если хотя бы один существует и конечен, то и другой существует и равен)
\end{lemma}

\begin{proof}
	\[
		\liml_{t \to +0} f(\vv{x}_0 + t \vv{d}) = l \Leftrightarrow (\forall \varepsilon > 0)(\exists \delta_1 > 0)(\forall t, 0 < t < \delta_1) \Leftrightarrow |f(\vv{x}_0 + t \vv{d}) - l| < \varepsilon
	\]
	В другом пределе будет то же за исключением условия на $\delta$, оно будет иметь вид: $0 < |t\vv{d}| < \delta \Leftrightarrow 0 < t < \frac{\delta}{|\vv{d}|}$(т.к. $t > 0$ изначально). БОО положив $\delta_1 = \frac{\delta}{|\vv{d}|}$ получим эквивалентность.
\end{proof}

\begin{theorem}
	Если существует предел по совокупности переменных $\liml_{\vv{x} \to \vv{x}_0} f(\vv{x}) = l$, то существует предел по любому направлению в точке $\vv{x}_0$, равный $l$. Обратное, вообще говоря, неверно.
\end{theorem}

\begin{proof}
	Само утверждение прямо вытекает из определений  пределов. Контрпримером обратного утверждения же будет $f(x, y)$:
	\[
		f(x, y) = \System{
			&{\frac{x^2 y}{x^4 + y^2},\ (x, y) \neq (0, 0)}
			\\
			&{0,\ (x, y) = (0, 0)}
		}
	\]
	Пусть $\vv{d} = (\alpha, \beta)$. Тогда
	\[
		D_{0, \vv{d}} = \{(x, y) \such x = \alpha t,\ y = \beta t,\ t \neq 0\}
	\]
	Посмотрим на предел $\liml_{t \to 0} f(\alpha t, \beta t)$:
	\[
		\liml_{t \to 0} f(\alpha t, \beta t) = \liml_{t \to 0} \frac{\alpha^2 \beta t^3}{\alpha^4 t^4 + \beta^2 t^2} = \liml_{t \to 0} \frac{\alpha^2 \beta t}{\alpha^4 t^2 + \beta^2} = 0
	\]
	То есть предел по любому направлению равен нулю. Однако предела по совокупности переменных нет: положим $x_k = \frac{1}{k},\ y_k = \frac{1}{k^2}$. В таком случае
	\[
		f\left(\frac{1}{k}, \frac{1}{k^2}\right) = \frac{1 / k^4}{2 / k^4} = \frac{1}{2}
	\]
\end{proof}

\begin{addition}
	На практике для проверки существования предела по совокупности используют следующий факт:
	
	Пусть есть $f(x, y)$, и нужно проверить $\liml_{(x, y) \to (0, 0)} f(x, y) = l$. Если, перейдя в полярные координаты и положив $x = \rho \cos \phi$, $y = \rho \sin \phi$, нам удастся доказать, что
	\[
		|f(\rho \cos\phi, \rho \sin\phi) - l| \le \psi(\rho),\ \liml_{\rho \to 0+} \psi(\rho) = 0
	\]
	где $\phi(\rho)$ - функция, зависящая только от $\rho$. Тогда действительно $\liml_{(x, y) \to (0, 0)} f(x, y) = l$
\end{addition}

\begin{definition}
	\textit{Повторным пределом} называется предел вида
	\[
		\liml_{x \to x_0} \left(\liml_{y \to y_0} f(x, y)\right)
	\]
\end{definition}

\begin{note}
	Вообще говоря, нет никакого общего соотношения между существованием повторного и двойного предела \textit{(смотри примеры)}.
\end{note}

\begin{example} 
	\textit{(из повторных не следует совокупность)}
	Рассмотрим функцию-контрпример из \textbf{Теоремы 1} данного параграфа. Легко видеть, что
	\[
		\liml_{x \to x_0} \left(\liml_{y \to y_0} f(x, y)\right) = \liml_{y \to y_0} \left(\liml_{x \to x_0} f(x, y)\right) = 0
	\]
	В то же время предела по совокупности, как мы уже убедились в доказательстве теоремы, у неё вообще нет.
\end{example}

\begin{example} 
	\textit{(из совокупности не следуют повторные)}
	Пусть $f(x, y)$ имеет вид:
	\[
		f(x, y) = \System{
			&{\sqrt{x^2 + y^2} \sin \frac{1}{x} \cos \frac{1}{y}},\ xy \neq 0
			\\
			&{0,\ xy = 0}
		}
	\]
	Заметим оценку:
	\[
		|f(\rho\cos\phi, \rho\sin\phi)| =  \left|\rho\sin\frac{1}{\rho\cos\phi}\cos\frac{1}{\rho\sin\phi}\right| \le \rho
	\]
	при $\cos\phi\sin\phi \neq 0$. То есть
	\[
		\liml_{(x, y) \to (0, 0)} f(x, y) = 0
	\]
	Но при этом повторного предела нет: если рассмотреть $\liml_{y \to 0} \left(\liml_{x \to 0} f(x, y)\right)$ и зафиксировать $\cos \frac{1}{y} \neq 0$, то получим
	\[
		\not\exists \liml_{x \to 0} f(x, y) = \liml_{x \to 0} \left(\sqrt{x^2 + y^2} \cos\frac{1}{y}\right)\sin\frac{1}{x}
	\]
\end{example}

\begin{definition}
	Если $\vv{x_0}$ - предельная точка области определения $D \subset \R^n$ функции $f: D \to \R$, тогда \textit{непрерывность в точке} $\vv{x_0}$ запишется в виде
	\[
		\liml_{\vv{x} \to \vv{x_0}} f(\vv{x}) = f(\vv{x_0})
	\]
	Если же $\vv{x_0}$ - не предельная, а изолированная точка области определения $D$ функции $f$, то тогда сразу $f$ непрерывна в $\vv{x}_0$ (так определим).
	Те же яйца, только в профиль, формы определений от наших любимцев:
	\begin{itemize}
		\item По Коши:
		\[
		(\forall \eps > 0)\ (\exists \delta > 0) : ( (\forall \vv{x} \in D,\ 0 < |\vv{x} - \vv{x_0}| < \delta\ )\hookrightarrow |f(\vv{x}) -f(\vv{x_0})| < \eps)
		\]
		
		\item По Гейне:
		\[
		\left(\forall \{\vv{x_m}\}_{m = 1}^\infty \subset D ,\ \liml_{m \to \infty} \vv{x_m} = \vv{x_0}\right) \hookrightarrow \liml_{m \to \infty} f(\vv{x_m}) = f(\vv{x_0})
		\]
	\end{itemize}
\end{definition}

\begin{note}
	Непрерывность по направлению определяется аналогично (в \textbf{обычном} определении просто все пределы по направлению, по \textbf{Коши} $\eps$-окрестности пересекать с лучом направления, а по \textbf{Гейне} последовательности Гейне брать только из луча направления(включая его начало). Остальные ограничения, естественно, никуда не уходят.
\end{note}

\begin{note}
	Такие свойства непрерывных функций, как замкнутость относительно сложения, вычитания, умножения, деления в точке, где знаменатель не равен нулю, доказываются аналогично доказательствам из прошлого семестра.
\end{note}

\begin{theorem} (Непрерывность сложной функции)
	Пусть $f: D \to \R$ непрерывна в точке $\vv{y}_0 = (y_1^{(0)}, \ldots, y_m^{(0)}) \in D \subset \R^m$, а также функции $\forall j \in \range{m}\ g_j \colon G \to \R$ непрерывны в точке $\vv{x}_0 = (x_1^{(0)}, \ldots, x_n^{(0)}) \in G \subset \R^n$, причём
	\begin{align*}
		&{\forall j \in \range{m}\ g_j(\vv{x}_0) = y_j^{(0)}}
		\\
		&{\forall \vv{x} \in G\ (g_1(\vv{x}), \ldots, g_m(\vv{x}))) \in D}
	\end{align*}
	Тогда сложная функция $h(\vv{x}) = f(g_1(\vv{x}), \ldots, g_m(\vv{x}))$ непрерывна в точке $\vv{x}_0$
\end{theorem}

\begin{proof}
	Рассмотрим последовательность Гейне $\forall \{\vv{x}_k\}_{k = 1}^\infty \subset G,\ \liml_{k \to \infty} \vv{x}_k = \vv{x}_0$. Тогда
	\[
		\forall j \in \range{m}\ \liml_{k \to \infty} g_j(\vv{x}_k) = g_j(\vv{x}_0) = y_j^{(0)}
	\]
	Если обозначить за $y_j^{(k)} := g_j(\vv{x}_k)$, то имеем:
	\[
		\vv{y}_k = (y_1^{(k)}, \ldots, y_m^{(k)}) \xrightarrow[k \to \infty]{} (y_1^{(0)}, \ldots, y_m^{(0)}) \lra \forall j \in \range{m}\ \liml_{k \to \infty} y_j^{(k)} = y_j^{(0)}
	\]
	В итоге получили, что
	\[
		\liml_{k \to \infty} f(\vv{y}_k) = f(\vv{y}_0) = \liml_{k \to \infty} h(\vv{x}_k) = h(\vv{x}_0)
	\]
\end{proof}

\begin{theorem}
	(Обобщение теоремы Вейерштрасса) Если $f$ непрерывна на компактном множестве $k \in \R^n$ то она ограничена и достигает на нём всех своих значений от $\inf$ до $\sup$ (множества значений) включительно. 
\end{theorem}

\begin{proof}
	Было проведено в прошлом семестре. А.Л. Лукашов предлагает заглянуть в
	\href{https://docs.360.yandex.ru/docs/view?url=ya-disk%3A%2F%2F%2Faa%2Fd_IDUn8-HtGvIBYA%2FLectures%2F1_Semester%2FCalculus%2F2023_Lukashov.pdf&name=2023_Lukashov.pdf&uid=1130000064610408}{конспект} (стр. 132).
\end{proof}

\begin{theorem} (Обобщение теоремы Кантора о равномерной непрерывности)
	Если $f$ непрерывна на компактном множестве $K \subset \R^n$, то она равномерно непрерывна на нём.
\end{theorem}

\begin{proof}
	От противного. В таком случае
	\[(0)\quad
		\exists \eps_0 > 0 \such \forall \delta > 0\ \exists \vv{x}_1, \vv{x}_2 \in K,\ |\vv{x}_1 - \vv{x}_2| < \delta\ \ |f(\vv{x}_1) - f(\vv{x}_2)| \ge \eps_0
	\]
	Последовательно для данного $\eps_0$ будем выбирать $\delta = 1, \frac{1}{2}, \ldots, \frac{1}{m}, \ldots$. Тогда имеем:
	\[(1)\quad
		\exists \vv{x}_{1, m}, \vv{x}_{2, m} \in K, |\vv{x}_{1, m} - \vv{x}_{2, m}| < \frac{1}{m} \quad |f(\vv{x}_{1, m}) - f(\vv{x}_{2, m})| \ge \eps_0
	\]
	Получили последовательность $\{\vv{x}_{1, m}\}_{m = 1}^\infty \subset K$, причём по обобщению теоремы Больцано-Вейерштрасса для компакта $K$ имеем, что
	\[(2)\quad
		\exists \{\vv{x}_{1, m_k}\}_{k = 1}^\infty,\ \liml_{k \to \infty} \vv{x}_{1, m_k} = \vv{x}_0 \in K
	\]
	Заметим несколько утверждений:
	\[
		\System{
			&{\text{По неравенству треугольника }|\vv{x}_{2, m_k} - \vv{x}_0| \le |\vv{x}_{2, m_k} - \vv{x}_{1, m_k}| + |\vv{x}_{1, m_k} - \vv{x}_0|}
			\\
			&{\text{Из (1) } |\vv{x}_{1, m} - \vv{x}_{2, m}| < \frac{1}{m} \to 0}
			\\
			&{\text{Из (2) } |\vv{x}_{1, m_k} - \vv{x}_0| \to 0}
		}
	\]
	Тогда 
	\[
		|\vv{x}_{2, m_k} - \vv{x}_0| \to 0 \lra \liml_{k \to \infty} \vv{x}_{2, m_k} = \vv{x}_0
	\]
	Но из непрерывности $f$ имеем:
	\[
		\liml_{k \to \infty} f(\vv{x}_{2, m_k}) = \liml_{k \to \infty} f(\vv{x}_{1, m_k}) = f(\vv{x}_0)
	\]
	А это противоречит (0).
\end{proof}

\begin{definition}
	Множество $A \subset E \subset \R^n$ называется \textit{относительно открытым (замкнутым)} в $E$, если $\exists$ открытое (замкнутое) множество $G \subset \R^n\ (F \subset \R^n)$ такое, что
	\[
		A = E \cap G\ (A = E \cap F)
	\]
\end{definition}

\begin{definition}
	Множество $E \subset \R^n$ (а вообще любое метрическое пространство) называется \textit{связным}, если его нельзя разбить на 2 непересекающихся непустых относительно открытых (замкнутых) в $E$ множеств.
\end{definition}

\begin{definition}
	Множество $E \subset \R^n$ называется \textit{линейно связным}, если $\forall \vv{x}_1, \vv{x}_2 \in E$ существует кривая $\gamma \subset E$, соединяющая $\vv{x}_1$ и $\vv{x}_2$.
\end{definition}
